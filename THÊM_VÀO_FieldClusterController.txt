    /**
     * ✅ API: Lấy availability + chi tiết bookings đã tồn tại
     * GET /api/public/field-clusters/{id}/availability-with-bookings?date=2026-04-25&field_type=5v5
     */
    public function getAvailabilityWithBookings(Request $request, string $id)
    {
        try {
            $validator = Validator::make($request->all(), [
                'date' => 'required|date',
                'field_type' => 'nullable|in:5v5,7v7,11v11',
            ]);

            if ($validator->fails()) {
                return ApiResponse::error('Validation failed', $validator->errors(), 422);
            }

            $cluster = FieldCluster::with(['fields'])->find($id);

            if (!$cluster) {
                return ApiResponse::error('Không tìm thấy cụm sân', null, 404);
            }

            $date = $request->date;
            $fieldType = $request->field_type;

            // Lấy các sân thuộc cluster (filter theo field_type nếu có)
            $fieldsQuery = Field::where('field_cluster_id', $cluster->id)
                ->where('status', 'available');

            if ($fieldType) {
                $fieldsQuery->where('field_type', $fieldType);
            }

            $fields = $fieldsQuery->get();

            // Xác định dayType (weekday/weekend)
            $dayOfWeek = Carbon::parse($date)->dayOfWeek;
            $dayType = in_array($dayOfWeek, [0, 6]) ? 'weekend' : 'weekday';

            // Array chứa kết quả
            $fieldsAvailability = [];

            foreach ($fields as $field) {
                $sessionsStatus = [];

                // Kiểm tra từng session
                foreach (['morning', 'afternoon', 'evening', 'full_day'] as $sessionType) {
                    [$startTime, $endTime] = $this->getSessionTimeRange($sessionType);

                    // ✅ Lấy TẤT CẢ bookings conflict với session này
                    $conflictBookings = Book::with(['user', 'tournament'])
                        ->where('field_id', $field->id)
                        ->where('day_book', $date)
                        ->where(function ($q) use ($startTime, $endTime) {
                            $q->whereBetween('start_time', [$startTime, $endTime])
                              ->orWhereBetween('end_time', [$startTime, $endTime])
                              ->orWhere(function ($query) use ($startTime, $endTime) {
                                  $query->where('start_time', '<=', $startTime)
                                        ->where('end_time', '>=', $endTime);
                              });
                        })
                        ->get();

                    $hasConflict = $conflictBookings->isNotEmpty();

                    // Lấy giá
                    $price = TournamentSessionPrice::where('field_cluster_id', $cluster->id)
                        ->where('field_type', $field->field_type)
                        ->where('day_type', $dayType)
                        ->where('session_type', $sessionType)
                        ->first();

                    // ✅ Format thông tin bookings đã tồn tại
                    $existingBookings = [];
                    foreach ($conflictBookings as $booking) {
                        $existingBookings[] = [
                            'id' => $booking->id,
                            'user' => [
                                'id' => $booking->user->id,
                                'name' => $booking->user->name,
                                'email' => $booking->user->email,
                            ],
                            'booking_type' => $booking->booking_type, // 'regular' hoặc 'tournament'
                            'tournament' => $booking->tournament ? [
                                'id' => $booking->tournament->id,
                                'name' => $booking->tournament->name,
                            ] : null,
                            'start_time' => $booking->start_time,
                            'end_time' => $booking->end_time,
                            'total_price' => $booking->total_price,
                        ];
                    }

                    $sessionsStatus[$sessionType] = [
                        'available' => !$hasConflict,
                        'time' => "$startTime-$endTime",
                        'price' => $price ? $price->price_per_session : null,
                        'reason' => $hasConflict ? "Đã có {$conflictBookings->count()} lịch đặt trong khung giờ này" : null,
                        'existing_bookings' => $existingBookings, // ✅ Thêm chi tiết bookings
                    ];
                }

                $fieldsAvailability[] = [
                    'field' => [
                        'id' => $field->id,
                        'name' => $field->name,
                        'field_type' => $field->field_type,
                        'status' => $field->status,
                    ],
                    'date' => $date,
                    'day_type' => $dayType,
                    'sessions' => $sessionsStatus,
                ];
            }

            return ApiResponse::success('Thông tin khả dụng các sân (kèm bookings)', [
                'field_cluster' => [
                    'id' => $cluster->id,
                    'name' => $cluster->name,
                    'address' => $cluster->address,
                ],
                'date' => $date,
                'day_type' => $dayType,
                'fields_availability' => $fieldsAvailability,
            ], 200);
        } catch (Exception $e) {
            return ApiResponse::error('Lấy thông tin thất bại', $e->getMessage(), 500);
        }
    }

    /**
     * Helper: Lấy time range của session
     */
    private function getSessionTimeRange(string $sessionType): array
    {
        return match ($sessionType) {
            'morning' => ['06:00:00', '11:00:00'],
            'afternoon' => ['13:00:00', '17:00:00'],
            'evening' => ['18:00:00', '22:00:00'],
            'full_day' => ['06:00:00', '22:00:00'],
            default => ['06:00:00', '22:00:00'],
        };
    }
